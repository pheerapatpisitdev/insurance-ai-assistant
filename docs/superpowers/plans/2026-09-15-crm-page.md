# CRM Page Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `/admin/crm` — one page showing how the bot is doing and who is worth calling back — reading the rows the recording layer now writes.

**Architecture:** The arithmetic lives in a pure module (`src/lib/crm/summary.ts`) that takes rows and returns numbers, so every figure on the page is unit-tested without a database. A server action fetches the rows and calls it. Customer names and pictures are fetched live from the Graph API at render and never stored.

**Tech Stack:** Next.js App Router Server Components, Tailwind v4, Supabase service-role client, Vitest.

**Spec:** `docs/superpowers/specs/2026-09-15-crm-dashboard-design.md` §7

**Depends on:** `docs/superpowers/plans/2026-09-15-crm-recording.md` (landed on `main`)

---

## Ground rules

1. **`requireAdmin()` goes inside the loader**, not only in `layout.tsx`. `loadAiPage()` is the
   counter-example this must not copy — it reads API-key tails with no guard of its own.
2. **The page must render with zero rows.** The tables are empty until the recording layer
   deploys, so the empty state is the first thing that will ever be seen, not an edge case.
3. **A Graph API failure must not blank the page.** A row with no name still shows its figures.
4. **Never store a name or a picture.** They are fetched per render for the rows on screen.
5. `npm run verify` must pass before every commit.

```ts
/** One conversation, as the page needs it. Mirrors the columns of `ins_conversations`. */
export interface ConversationRow {
  id: string;
  started_at: string;
  last_event_at: string | null;
  product: string | null;
  source: string | null;
  ad_id: string | null;
  ref: string | null;
  priced_at: string | null;
  form_sent_at: string | null;
  form_done_at: string | null;
  agent_replied_at: string | null;
  stalled_at: string | null;
  handover_at: string | null;
  messages: number | null;
}

/** One lead, as the page needs it. Mirrors the columns of `ins_leads`. */
export interface LeadRow {
  id: string;
  conversation_id: string;
  stage: string;
  product: string | null;
  last_quote: Record<string, unknown> | null;
  ad_id: string | null;
  created_at: string;
  updated_at: string;
  has_psid: boolean;
}

/** A question the bot had no answer for. */
export interface UnansweredRow {
  id: number;
  at: string;
  product: string | null;
  intent: string | null;
  question: string;
}

/** How far back the page is looking. */
export type Range = "today" | "7d" | "30d";

export interface Counts {
  arrived: number;
  told: number;
  priced: number;
  interested: number;
  formDone: number;
  stalled: number;
  agentReplied: number;
}

export interface Summary {
  counts: Counts;
  /** the same counts again, split by product */
  byProduct: { product: string; counts: Counts }[];
  /** one entry per day in the range, oldest first */
  byDay: { date: string; arrived: number; priced: number; interested: number }[];
  /** 24 entries, midnight to midnight, in the page's own timezone */
  byHour: { hour: number; arrived: number }[];
  /** the advertisements that brought anyone, busiest first */
  byAd: { adId: string; arrived: number; interested: number }[];
}
```

- [ ] **Step 2: Write the failing tests**

Create `tests/admin/crm-summary.test.ts`:

```ts
import { describe, expect, it } from "vitest";
import { rangeStart, summarise } from "@/lib/crm/summary";
import type { ConversationRow } from "@/lib/crm/types";

/** A conversation that arrived at `at` and got no further than `reached`. */
function conv(id: string, at: string, reached: Partial<ConversationRow> = {}): ConversationRow {
  return {
    id, started_at: at, last_event_at: at, product: "lifeprotect",
    source: "organic", ad_id: null, ref: null,
    priced_at: null, form_sent_at: null, form_done_at: null,
    agent_replied_at: null, stalled_at: null, handover_at: null,
    messages: 1, ...reached,
  };
}

const D1 = "2026-09-14T03:00:00.000Z";
const D2 = "2026-09-15T04:00:00.000Z";

describe("counting what happened", () => {
  it("counts nothing without pretending there is something", () => {
    const s = summarise([], [], "7d", new Date(D2));
    expect(s.counts).toEqual({
      arrived: 0, told: 0, priced: 0, interested: 0, formDone: 0, stalled: 0, agentReplied: 0,
    });
    expect(s.byDay).toHaveLength(7);
    expect(s.byDay.every((d) => d.arrived === 0)).toBe(true);
  });

  it("counts an arrival for every conversation", () => {
    const s = summarise([conv("a", D2), conv("b", D2)], [], "7d", new Date(D2));
    expect(s.counts.arrived).toBe(2);
  });

  it("counts a quotation only where one was stamped", () => {
    const s = summarise([conv("a", D2, { priced_at: D2 }), conv("b", D2)], [], "7d", new Date(D2));
    expect(s.counts.priced).toBe(1);
  });

  it("counts someone who asked to apply, by the milestone and not by the lead", () => {
    const s = summarise([conv("a", D2, { handover_at: D2 })], [], "7d", new Date(D2));
    expect(s.counts.interested).toBe(1);
  });

  it("counts a conversation that reached the form all the way through", () => {
    const rows = [conv("a", D2, { priced_at: D2, form_sent_at: D2, form_done_at: D2, handover_at: D2 })];
    const s = summarise(rows, [], "7d", new Date(D2));
    expect(s.counts).toMatchObject({ arrived: 1, priced: 1, interested: 1, formDone: 1 });
  });

  it("counts a person the bot answered but never priced as told, not priced", () => {
    const s = summarise([conv("a", D2, { messages: 4 })], [], "7d", new Date(D2));
    expect(s.counts.told).toBe(1);
    expect(s.counts.priced).toBe(0);
  });
});

describe("splitting by plan", () => {
  it("keeps each plan's figures apart", () => {
    const rows = [
      conv("a", D2, { product: "lifeprotect", priced_at: D2 }),
      conv("b", D2, { product: "ihealthy" }),
      conv("c", D2, { product: "ihealthy", priced_at: D2, handover_at: D2 }),
    ];
    const s = summarise(rows, [], "7d", new Date(D2));
    const life = s.byProduct.find((p) => p.product === "lifeprotect")!;
    const health = s.byProduct.find((p) => p.product === "ihealthy")!;
    expect(life.counts).toMatchObject({ arrived: 1, priced: 1, interested: 0 });
    expect(health.counts).toMatchObject({ arrived: 2, priced: 1, interested: 1 });
  });

  it("gives a conversation with no plan chosen a bucket of its own rather than dropping it", () => {
    const s = summarise([conv("a", D2, { product: null })], [], "7d", new Date(D2));
    expect(s.counts.arrived).toBe(1);
    expect(s.byProduct.reduce((n, p) => n + p.counts.arrived, 0)).toBe(1);
  });
});

describe("laying the days out", () => {
  it("gives every day in the range a slot, including the ones nobody wrote on", () => {
    const s = summarise([conv("a", D1), conv("b", D2), conv("c", D2)], [], "7d", new Date(D2));
    expect(s.byDay).toHaveLength(7);
    expect(s.byDay.at(-1)).toMatchObject({ arrived: 2 });
    expect(s.byDay.at(-2)).toMatchObject({ arrived: 1 });
    expect(s.byDay.slice(0, 5).every((d) => d.arrived === 0)).toBe(true);
  });

  it("leaves out what happened before the range began", () => {
    const old = "2026-08-01T00:00:00.000Z";
    const s = summarise([conv("a", old), conv("b", D2)], [], "7d", new Date(D2));
    expect(s.counts.arrived).toBe(1);
  });

  it("gives all twenty-four hours a slot", () => {
    const s = summarise([conv("a", D2)], [], "7d", new Date(D2));
    expect(s.byHour).toHaveLength(24);
    expect(s.byHour.reduce((n, h) => n + h.arrived, 0)).toBe(1);
  });
});

describe("which advertisement brought them", () => {
  it("groups by ad and puts the busiest first", () => {
    const rows = [
      conv("a", D2, { ad_id: "ad-1" }),
      conv("b", D2, { ad_id: "ad-1", handover_at: D2 }),
      conv("c", D2, { ad_id: "ad-2" }),
    ];
    const s = summarise(rows, [], "7d", new Date(D2));
    expect(s.byAd[0]).toEqual({ adId: "ad-1", arrived: 2, interested: 1 });
    expect(s.byAd[1]).toEqual({ adId: "ad-2", arrived: 1, interested: 0 });
  });

  it("leaves out those who arrived by no advertisement at all", () => {
    const s = summarise([conv("a", D2)], [], "7d", new Date(D2));
    expect(s.byAd).toHaveLength(0);
  });
});

describe("where a range begins", () => {
  it("starts today at this morning, not twenty-four hours ago", () => {
    const now = new Date("2026-09-15T16:30:00.000Z");
    const start = rangeStart("today", now);
    expect(start.getTime()).toBeLessThan(now.getTime());
    expect(now.getTime() - start.getTime()).toBeLessThan(24 * 3600_000);
  });

  it("counts seven days back for the week and thirty for the month", () => {
    const now = new Date(D2);
    const week = now.getTime() - rangeStart("7d", now).getTime();
    const month = now.getTime() - rangeStart("30d", now).getTime();
    expect(Math.round(week / 86_400_000)).toBeGreaterThanOrEqual(6);
    expect(Math.round(month / 86_400_000)).toBeGreaterThanOrEqual(29);
  });
});
```

- [ ] **Step 4: Write the implementation**

Create `src/lib/crm/summary.ts` implementing `rangeStart(range, now)` and
`summarise(conversations, leads, range, now)` against the types above. Requirements the tests
pin down:

- `told` counts a conversation with `messages >= 2` — the bot said something back
- `priced`, `interested`, `formDone`, `stalled`, `agentReplied` each count a non-null milestone
- `byProduct` buckets a null product under `"undecided"` rather than dropping the row
- `byDay` emits one entry per day of the range, zero-filled, oldest first
- `byHour` emits all 24 hours, zero-filled
- `byAd` includes only rows with an `ad_id`, sorted by `arrived` descending
- rows started before `rangeStart` are excluded from every figure
